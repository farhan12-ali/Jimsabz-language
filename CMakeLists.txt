cmake_minimum_required(VERSION 3.16)
project(Jimsabz VERSION 0.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(jimsabz src/gc.cpp src/types.cpp src/elaborate.cpp src/standard.cpp src/verify.cpp src/frontend.cpp src/semantic.cpp src/compiler.cpp src/bytecode.cpp
    src/vm.cpp src/graphics.cpp src/network.cpp src/platform.cpp src/studio_win32.cpp src/main.cpp)
target_include_directories(jimsabz PRIVATE include)
if(MSVC)
    set_property(TARGET jimsabz PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(jimsabz PRIVATE /utf-8 /W3)
endif()
if(WIN32)
    if(MINGW)
        target_compile_options(jimsabz PRIVATE -pthread)
        target_link_options(jimsabz PRIVATE -static -pthread)
    endif()
    target_compile_definitions(jimsabz PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(jimsabz PRIVATE user32 gdi32 comdlg32 comctl32 ole32 uuid windowscodecs winmm sapi ws2_32 shell32)
    add_executable(Jimsabz-Studio WIN32 src/gc.cpp src/types.cpp src/elaborate.cpp src/standard.cpp src/verify.cpp src/frontend.cpp src/semantic.cpp src/compiler.cpp src/bytecode.cpp
        src/vm.cpp src/graphics.cpp src/network.cpp src/platform.cpp src/studio_win32.cpp src/studio_entry.cpp)
    target_include_directories(Jimsabz-Studio PRIVATE include)
    target_compile_definitions(Jimsabz-Studio PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(Jimsabz-Studio PRIVATE user32 gdi32 comdlg32 comctl32 ole32 uuid windowscodecs winmm sapi ws2_32 shell32)
    if(MSVC)
        set_property(TARGET Jimsabz-Studio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_options(Jimsabz-Studio PRIVATE /utf-8 /W3)
    elseif(MINGW)
        target_compile_options(Jimsabz-Studio PRIVATE -pthread)
        target_link_options(Jimsabz-Studio PRIVATE -municode -static -pthread)
    endif()
else()
    find_package(Threads REQUIRED)
    target_link_libraries(jimsabz PRIVATE Threads::Threads)
endif()
